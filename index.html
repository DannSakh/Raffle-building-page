// Fisher-Yates shuffle algorithm function shuffle(arr) { const shuffled = [...arr]; for (let i = shuffled.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]]; } return shuffled; } function highlightWinnerSegment(segmentIndex) { const segments = wheel.querySelectorAll('.wheel-segment'); if (segments[segmentIndex]) { segments[segmentIndex].style.filter = 'brightness(1.5) saturate(1.5)'; segments[segmentIndex].style.boxShadow = '0 0 30px gold inset'; segments[segmentIndex].style.transition = 'all 0.5s ease'; } } function spinWheel() { if (isSpinning || participants.length === 0) return; if (usedParticipants.length >= WINNERS_PER_ROUND) { showStatus('🎉 All prizes have already been awarded. Please reset to start again.', 'error'); spinBtn.disabled = true; return; } const availableParticipants = participants.filter(p => !usedParticipants.includes(p)); if (availableParticipants.length < WINNERS_PER_ROUND) { showStatus(`⚠️ Not enough participants remaining. Need at least ${WINNERS_PER_ROUND}.`, 'error'); return; } isSpinning = true; spinBtn.disabled = true; wheel.classList.add('spinning'); const baseSpins = 5 + Math.floor(Math.random() * 3); const extraDegrees = Math.random() * 360; const totalRotation = (baseSpins * 360) + extraDegrees; const currentTransform = wheel.style.transform; const currentRotation = currentTransform ? parseFloat(currentTransform.replace('rotate(', '').replace('deg)', '')) : 0; const finalRotation = currentRotation + totalRotation; wheel.style.transition = 'transform 4s cubic-bezier(0.17, 0.67, 0.12, 0.99)'; wheel.style.transform = `rotate(${finalRotation}deg)`; setTimeout(() => { // Calculate which segment is at the top (0 degrees) const normalizedAngle = ((finalRotation % 360) + 360) % 360; const segmentAngle = 360 / availableParticipants.length; // Winner 1 is determined by wheel position const winner1Index = Math.floor(normalizedAngle / segmentAngle); const winner1 = availableParticipants[winner1Index]; // Highlight the winning segment highlightWinnerSegment(winner1Index); // Winners 2 and 3 are chosen using Fisher-Yates from remaining participants const remaining = availableParticipants.filter((p, i) => i !== winner1Index); const shuffled = shuffle(remaining); const winners = [ { participant: winner1, prize: PRIZES[0] }, { participant: shuffled[0], prize: PRIZES[1] }, { participant: shuffled[1], prize: PRIZES[2] } ]; // Mark all winners as used winners.forEach(w => usedParticipants.push(w.participant)); totalWinnersCount += winners.length; switchToAfterSpinLayout(); displayWinners(winners); isSpinning = false; wheel.classList.remove('spinning'); resetBtn.disabled = false; updateWinnersInfo(); if (usedParticipants.length >= WINNERS_PER_ROUND) { spinBtn.disabled = true; showStatus(`🎉 ${WINNERS_PER_ROUND} winners selected! All prizes awarded.`, 'success'); } else { spinBtn.disabled = false; showStatus(`🎉 ${winners.length} winners selected!`, 'success'); } }, 4000); }
